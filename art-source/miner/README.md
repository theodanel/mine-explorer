## Quick start

```js
// Per-direction character — load one atlas per group, then create animations.
import Phaser from 'phaser'

class CharacterScene extends Phaser.Scene {
  preload() {
    this.load.json('spriteship-export', 'tizo-export.json')
    this.load.atlas("Petit mineur dragon aventurier-pickaxe_swing", "spritesheet_pickaxe_swing.png", "atlas_pickaxe_swing.json")
  }

  create() {
    const payload = this.cache.json.get('spriteship-export').payload
    for (const group of payload.groups) {
      const textureKey = "Petit mineur dragon aventurier-" + group.key
      for (const [name, clip] of Object.entries(group.atlas.animations)) {
        this.anims.create({ key: name,
          frames: clip.frames.map(frame => ({ key: textureKey, frame })),
          frameRate: clip.fps, repeat: clip.loop ? -1 : 0 })
      }
    }
    const sprite = this.add.sprite(200, 200, "Petit mineur dragon aventurier-pickaxe_swing", "pickaxe_swing_0000")
    sprite.anims.play("pickaxe_swing")
  }
}
```
